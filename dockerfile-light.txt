# Build stage
FROM alpine:3.11 as builder

# Install build dependencies
RUN apk add --no-cache \
    python2 \
    py2-pip \
    nodejs \
    npm \
    gcc \
    python2-dev \
    musl-dev \
    linux-headers

# Set working directory
WORKDIR /app

# Copy only files needed for building
COPY flask-app/requirements.txt .
COPY flask-app/package.json .
COPY flask-app/package-lock.json .
COPY flask-app/webpack.config.js .
COPY flask-app/.babelrc .
COPY flask-app/src ./src/

# Install dependencies and build frontend
RUN npm install
RUN npm run build

# Final stage
FROM alpine:3.11

# Install only runtime dependencies
RUN apk add --no-cache \
    python2 \
    py2-pip

# Set working directory
WORKDIR /app

# Copy Python requirements and install
COPY flask-app/requirements.txt .
RUN pip install -r requirements.txt

# Copy application files
COPY flask-app/app.py .
COPY flask-app/templates ./templates/

# Copy built static files from builder
COPY --from=builder /app/static/build ./static/build/

EXPOSE 5000

ENTRYPOINT ["python", "app.py"]
